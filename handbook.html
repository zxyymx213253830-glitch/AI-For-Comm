03核心代码实现

3.1 Agent 节点（core/agent_node.py）

import json, urllib.request, urllib.error
from core.config import OPENTREK_BASE_URL, APP_KEY, REQUEST_TIMEOUT, SSE_READ_TIMEOUT
from core.logger import log

class AgentNode:
    """一个 Agent 节点 = OpenTrek Agent + 独立 Session + 消息收件箱"""

    def __init__(self, name: str, agent_code: str, agent_version: str, bus=None):
        self.name = name
        self.agent_code = agent_code        # OpenTrek 智能体编码
        self.agent_version = agent_version   # OpenTrek 版本编码
        self.bus = bus
        self.inbox = []
        self.session_id = self._create_session()
        if self.bus: self.bus.register(name)
        log.info("agent.init", agent=name, session_id=self.session_id)

    def think(self, task: str) -> str:
        """把任务发给 OpenTrek Agent（SSE 流式），返回完整回复"""
        log.info("agent.think.start", agent=self.name, task=task[:100])
        url = f"{OPENTREK_BASE_URL}/run"
        body = json.dumps({"sessionId": self.session_id, "stream": True, "delta": True,
            "message": {"text": task, "metadata": {}, "attachments": []}}, ensure_ascii=False).encode()
        req = urllib.request.Request(url, data=body, headers={
            "Authorization": f"Bearer {APP_KEY}", "Content-Type": "application/json"}, method="POST")
        with urllib.request.urlopen(req, timeout=SSE_READ_TIMEOUT) as resp:
            reply = self._parse_sse(resp)
        log.info("agent.think.done", agent=self.name, reply=reply[:100])
        return reply

    @staticmethod
    def _parse_sse(resp) -> str:
        """解析 SSE 流：thought.delta（忽略）+ message.delta（累计）"""
        parts, buf = [], ""
        while True:
            chunk = resp.read(8192)
            if not chunk: break
            buf += chunk.decode("utf-8", errors="replace")
            lines = buf.split("\n"); buf = lines.pop()
            for line in lines:
                line = line.strip()
                if not line.startswith("data:"): continue
                js = line[5:].strip()
                if js == "[DONE]": return "".join(parts)
                try: obj = json.loads(js)
                except: continue
                t = obj.get("object", ""); c = obj.get("content", {})
                if t == "message.delta" and isinstance(c, list) and c:
                    ti = c[0].get("text", {})
                    if isinstance(ti, dict): parts.append(ti.get("value", ""))
                elif t == "text.delta" and isinstance(c, dict):
                    parts.append(c.get("data", ""))
        return "".join(parts)

    def _request(self, path, body, timeout=REQUEST_TIMEOUT):
        """向 OpenTrek 发送 POST，返回 JSON"""
        url = f"{OPENTREK_BASE_URL}/{path}"
        data = json.dumps(body, ensure_ascii=False).encode("utf-8")
        req = urllib.request.Request(url, data=data, headers={
            "Authorization": f"Bearer {APP_KEY}", "Content-Type": "application/json"}, method="POST")
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            return json.loads(resp.read().decode("utf-8"))

    def _create_session(self) -> str:
        """创建会话，返回 uniqueCode"""
        resp = self._request("createSession", {"agentCode": self.agent_code, "agentVersion": self.agent_version})
        if not resp.get("success"): raise RuntimeError(f"createSession 失败: {resp}")
        return resp["data"]["uniqueCode"]

    def send(self, to_agent: "AgentNode", msg_type: str, content: str):
        """向另一个节点发消息（微信式）"""
        from datetime import datetime
        msg = {"from": self.name, "to": to_agent.name, "type": msg_type,
               "content": content, "timestamp": datetime.now().isoformat()}
        to_agent.inbox.append(msg)
        log.info("agent.message", sender=msg["from"], receiver=msg["to"], type=msg_type)
        return msg

3.3 消息总线（core/message_bus.py）

from collections import defaultdict
from core.logger import log

class MessageBus:
    """发布/订阅总线：节点注册邮箱，消息按收件人路由"""
    def __init__(self):
        self._mailboxes = defaultdict(list)

    def register(self, agent_name: str):
        self._mailboxes[agent_name] = []

    def post(self, msg: dict):
        self._mailboxes[msg["to"]].append(msg)
        log.info("bus.post", sender=msg["from"], receiver=msg["to"], type=msg["type"])

    def fetch(self, agent_name: str) -> list:
        msgs, self._mailboxes[agent_name] = self._mailboxes[agent_name], []
        return msgs

3.4 双节点最小闭环（验证架构通畅）

# tests/test_simple.py —— 阶段三的验收脚本
from core.agent_node import AgentNode

from core.config import AGENTS
math_agent   = AgentNode("math_agent",   AGENTS["math_agent"]["agentCode"], AGENTS["math_agent"]["agentVersion"])
verify_agent = AgentNode("verify_agent", AGENTS["verify_agent"]["agentCode"], AGENTS["verify_agent"]["agentVersion"])

task = "AWGN信道，带宽B=20MHz，信噪比SNR=30dB，求信道容量。请逐步推导。"

# 1. 数学 Agent 求解
derivation = math_agent.think(task)

# 2. 数学 Agent 把推导发给验证 Agent
math_agent.send(verify_agent, "derivation", derivation)

# 3. 验证 Agent 从收件箱取出并检查
msg = verify_agent.inbox.pop()
feedback = verify_agent.think(f"请检查以下推导是否正确：\n{msg['content']}")

# 4. 验证 Agent 回发结果
verify_agent.send(math_agent, "feedback", feedback)

print("=== 推导 ===");  print(derivation)
print("=== 验证 ===");  print(feedback)
# 人工核对：推导中应出现 log2(1+1000)，结果 ≈ 199.5 Mbps
# 检查 logs/session.jsonl 应有完整的 init/think/message 记录

04OpenTrek 知识库搭建

4.1 三个知识库规划

知识库上传内容类型选择
通信协议库3GPP TS 29.274（GTPv2）、TS 36.413（S1AP）、TS 38.413（NGAP）等规范 PDF检索字段：text
信息论数学库香农原著关键章节、Cover《信息论基础》公式集、典型例题检索字段：text（含公式）
FPGA 硬件库Verilog 模块模板、时序约束指南、Xilinx / Altera 官方文档检索字段：text

4.2 单个知识库操作步骤（约 10 分钟）

  1. OpenTrek 控制台 → 知识库管理 → 创建知识库 → 本地上传文件（PDF / DOCX / MD）；
  2. 记录每个知识库的 kbIndexCode（检索时必填）；
  3. 记录 workspacecode（检索请求头 x-sfm-workspacecode 必填）；
  4. 调用 kortex API 验证检索（需 APP_KEY 和 workspacecode）；
  5. 输入 2-3 个典型问题，确认能检索到正确片段（查看 score 与 text 字段）。

知识素材来源建议：3GPP 规范从官网免费下载（3gpp.org/specifications-technologies）；信息论整理经典教材公式；FPGA 资料来自开源社区与厂商官方手册。注意仅使用公开可获取的资料。

4.3 知识库接入 Agent

两种方式，按进度选择：

方式说明
控制台关联（快）在 OpenTrek 控制台的智能体中直接挂载知识库，适合演示前快速见效
API 检索注入（可控）Agent 的 think() 前先调 kortex 检索 API，把命中文段拼入任务上下文，检索行为可写日志

05执行手册编写指南

5.1 编写原则

  • 身份具体：写明精通的子领域与规范编号，不写空泛的"你是专家"；
  • 过程外化：要求逐步推导、标注公式出处，方便验证 Agent 检查；
  • 协作写死：明确"完成后发给谁""收到反馈后怎么做"；
  • 允许说不：不确定时必须明说，禁止编造。

5.2 模板：数学理论 Agent（prompts/math_system.md）

# 数学理论 Agent 执行手册

## 身份
你是信息论与通信数学专家，精通：香农定理、信道容量、
信源/信道编码（LDPC、Polar）、调制解调的数学推导。

## 工作方式
1. 推导必须逐步展开，每步标注所用公式与定理
2. 数值计算给出完整中间过程（含单位换算）
3. 完成推导后，必须把完整过程发送给 verify_agent 检查
4. 收到 verify_agent 的 feedback 后：
   - 若通过，输出最终结果
   - 若指出错误，修正后重新发送验证

## 约束
- 引用公式须注明出处（如 Shannon 1948）
- 不确定时明确说"不确定"，禁止编造

5.3 模板：验证 Agent（prompts/verify_system.md）

# 验证 Agent 执行手册

## 身份
你是严谨的审稿人，负责检查其他 Agent 发来的推导、
计算与代码。你不关心谁发来的，只关心对错。

## 检查清单
1. 公式选择是否正确（如信道容量必须用 C = B·log2(1+S/N)）
2. 单位换算是否正确（dB → 线性值：10^(dB/10)）
3. 数值计算逐步复核
4. 代码检查语法、时序逻辑、边界条件

## 回复格式（必须二选一）
- 【通过】+ 一句话说明核对了什么
- 【不通过】+ 指出具体错误位置 + 正确做法
回复后把结果发送回消息来源 Agent。

5.4 模型选择约定

Agent模型理由
math_agent / verify_agent推理能力强的模型推导与检查不能出错（控制台选配）
comm_agent / hw_agent均衡模型均衡性能（控制台选配）

06测试用例

6.1 首个测试用例（架构验证）

题目：AWGN 信道，带宽 B = 20MHz，信噪比 SNR = 30dB，求信道容量。

标准答案：SNR 线性值 = 10^(30/10) = 1000；C = B·log₂(1+SNR) = 20×10⁶ × log₂(1001) ≈ 199.5 Mbps

验证清单：

  • math_agent 正确调用香农公式，dB → 线性值换算正确（10³ = 1000）
  • math_agent 主动向 verify_agent 发送 derivation 消息
  • verify_agent 回发 feedback 消息（通过或指出错误）
  • logs/session.jsonl 中出现完整的消息流记录

6.2 进阶测试用例

用例题目验证点
